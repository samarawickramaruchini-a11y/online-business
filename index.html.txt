        // PRODUCT DATA
        const products = [

            {

                tag: "Relaxing",
                imagePlaceholder: "🧘"
            },
            {
                id: 4,
                name: "Rainbow Plaster Set",
                price: 15.00,
                category: "kids",
                tag: "Sale",
                imagePlaceholder: "🌈"
            },
            {
                id: 5,
                name: "Home Decor Craft Box",
                price: 45.00,
                category: "gypsum",
                tag: "Premium",
                imagePlaceholder: "🏠"
            },
            {
                id: 6,
                name: "Beginner Painting Palette",
                price: 12.99,
                category: "adult",
                tag: "Essential",
                imagePlaceholder: "🎨"
            }
        ];

        // STATE
        let cart = JSON.parse(localStorage.getItem('nuvora-cart')) || [];

        // DOM ELEMENTS
        const productList = document.getElementById('product-list');
        const cartCount = document.getElementById('cart-count');
        const cartBtn = document.getElementById('cart-btn');
        const cartDrawer = document.getElementById('cart-drawer');
        const closeCart = document.getElementById('close-cart');
        const overlay = document.getElementById('overlay');
        const cartItemsList = document.getElementById('cart-items-list');
        const cartTotal = document.getElementById('cart-total');
        const filterBtns = document.querySelectorAll('.filter-btn');

        // INITIALIZE
        function init() {
            renderProducts(products);
            updateCartUI();
        }

        // RENDER PRODUCTS
        function renderProducts(items) {
            productList.innerHTML = '';
            items.forEach(product => {
                const card = document.createElement('div');
                card.className = 'product-card';
                card.innerHTML = `
                    <div class="product-tag">${product.tag}</div>
                    <div class="product-image" style="font-size: 4rem;">${product.imagePlaceholder}</div>
                    <div class="product-info">
                        <h3>${product.name}</h3>
                        <span class="product-price">$${product.price.toFixed(2)}</span>
                        <button class="btn btn--primary" style="width: 100%; font-size: 0.7rem; padding: 8px;" onclick="addToCart(${product.id})">Add to Cart</button>
                    </div>
                `;
                productList.appendChild(card);
            });
        }

        // FILTERING
        filterBtns.forEach(btn => {
            btn.addEventListener('click', () => {
                filterBtns.forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                const filter = btn.dataset.filter;
                const filtered = filter === 'all' ? products : products.filter(p => p.category === filter);
                renderProducts(filtered);
            });
        });

        // CART LOGIC
        window.addToCart = function(id) {
            const product = products.find(p => p.id === id);
            const existing = cart.find(item => item.id === id);
            
            if (existing) {
                existing.quantity += 1;
            } else {
                cart.push({...product, quantity: 1});
            }
            
            saveCart();
            updateCartUI();
            openCart();
        };

        function removeFromCart(id) {
            cart = cart.filter(item => item.id !== id);
            saveCart();
            updateCartUI();
        }

        function saveCart() {
            localStorage.setItem('nuvora-cart', JSON.stringify(cart));
        }

        function updateCartUI() {
            // Count
            const totalItems = cart.reduce((sum, item) => sum + item.quantity, 0);
            cartCount.textContent = totalItems;

            // List
            cartItemsList.innerHTML = '';
            let total = 0;

            if(cart.length === 0) {
                cartItemsList.innerHTML = '<p style="text-align:center; margin-top: 20px; opacity: 0.5;">Your cart is empty.</p>';
            }

            cart.forEach(item => {
                const itemEl = document.createElement('div');
                itemEl.className = 'cart-item';
                itemEl.innerHTML = `
                    <div style="background: var(--cream); width: 60px; height: 60px; display: flex; align-items: center; justify-content: center; border-radius: 8px; font-size: 1.5rem;">${item.imagePlaceholder}</div>
                    <div style="flex-grow: 1;">
                        <h4 style="font-size: 0.9rem;">${item.name}</h4>
                        <div style="font-size: 0.8rem; opacity: 0.7;">$${item.price} x ${item.quantity}</div>
                    </div>
                    <div style="cursor: pointer; color: var(--coral);" onclick="removeFromCart(${item.id})">Remove</div>
                `;
                cartItemsList.appendChild(itemEl);
                total += item.price * item.quantity;
            });

            cartTotal.textContent = `$${total.toFixed(2)}`;
        }

        // DRAWER CONTROLS
        function openCart() {
            cartDrawer.classList.add('open');
            overlay.classList.add('active');
        }

        function closeCartDrawer() {
            cartDrawer.classList.remove('open');
            overlay.classList.remove('active');
        }

        cartBtn.addEventListener('click', openCart);
        closeCart.addEventListener('click', closeCartDrawer);
        overlay.addEventListener('click', closeCartDrawer);

        // NEWSLETTER
        document.getElementById('newsletter-form').addEventListener('submit', (e) => {
            e.preventDefault();
            alert('Thank you for subscribing to NUVORA! This is a demo - no email was sent.');
            e.target.reset();
        });

        // Start
        init();

    </script>
